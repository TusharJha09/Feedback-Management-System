import java.math.BigDecimal; import java.math.BigInteger; import java.math.MathContext; import java.math.RoundingMode; import java.util.ArrayList; import java.util.List; import java.util.Locale; import java.util.Scanner; public class Calculator { private static final Scanner SCANNER = new Scanner(System.in); private static final MathContext MC = new MathContext(34, RoundingMode.HALF_EVEN); private static final int MAX_FACTORIAL = 1000; private static final int MAX_INTEGER_POWER = 10000; private static final List HISTORY = new ArrayList<>(); private static BigDecimal memory = BigDecimal.ZERO; private static boolean degreeMode = true; public static void main(String[] args) { System.out.println("========================================"); System.out.println(" ADVANCED JAVA CALCULATOR"); System.out.println("========================================"); try { while (true) { displayMenu(); System.out.print("\nChoose an option: "); if (!SCANNER.hasNextLine()) { break; } String choice = SCANNER.nextLine() .trim().toLowerCase(Locale.ROOT); if (choice.equals("exit") || choice.equals("0")) { break; } try { if (!processChoice(choice)) { break; } } catch (ArithmeticException | IllegalArgumentException e) { System.out.println("Error: " + e.getMessage()); } } } finally { SCANNER.close(); } System.out.println("\nCalculator closed. Goodbye!"); } // ---------------- MENU ---------------- private static void displayMenu() { System.out.println("\n--------------- MENU ----------------"); System.out.println(" BASIC OPERATIONS"); System.out.println(" 1. Addition (+)"); System.out.println(" 2. Subtraction (-)"); System.out.println(" 3. Multiplication (*)"); System.out.println(" 4. Division (/)"); System.out.println(" 5. Modulus (%)"); System.out.println(" 6. Power (^)"); System.out.println(" 7. Percentage"); System.out.println(); System.out.println(" SCIENTIFIC OPERATIONS"); System.out.println(" 8. Square root"); System.out.println(" 9. Factorial"); System.out.println("10. Sine"); System.out.println("11. Cosine"); System.out.println("12. Tangent"); System.out.println("13. Log base 10"); System.out.println("14. Natural logarithm"); System.out.println("15. Exponential (e^x)"); System.out.println("16. Absolute value"); System.out.println(); System.out.println(" MEMORY AND UTILITIES"); System.out.println("17. Store result in memory (MS)"); System.out.println("18. Recall memory (MR)"); System.out.println("19. Add value to memory (M+)"); System.out.println("20. Clear memory (MC)"); System.out.println("21. View calculation history"); System.out.println("22. Clear calculation history"); System.out.println("23. Toggle degree/radian mode"); System.out.println(" 0. Exit"); System.out.println("-------------------------------------"); System.out.println("Angle mode: " + (degreeMode ? "DEGREES" : "RADIANS")); System.out.println("Memory: " + format(memory)); } // ---------------- MENU CONTROLLER ---------------- private static boolean processChoice(String choice) { switch (choice) { case "1": binaryOperation("+"); break; case "2": binaryOperation("-"); break; case "3": binaryOperation("*"); break; case "4": binaryOperation("/"); break; case "5": binaryOperation("%"); break; case "6": binaryOperation("^"); break; case "7": binaryOperation("percent"); break; case "8": unaryOperation("sqrt"); break; case "9": unaryOperation("factorial"); break; case "10": unaryOperation("sin"); break; case "11": unaryOperation("cos"); break; case "12": unaryOperation("tan"); break; case "13": unaryOperation("log"); break; case "14": unaryOperation("ln"); break; case "15": unaryOperation("exp"); break; case "16": unaryOperation("abs"); break; case "17": storeMemory(); break; case "18": recallMemory(); break; case "19": addToMemory(); break; case "20": memory = BigDecimal.ZERO; System.out.println("Memory cleared."); break; case "21": showHistory(); break; case "22": HISTORY.clear(); System.out.println("Calculation history cleared."); break; case "23": degreeMode = !degreeMode; System.out.println("Angle mode changed to " + (degreeMode ? "DEGREES" : "RADIANS")); break; default: System.out.println("Invalid choice. Select an option from 0 to 23."); } return true; } // ---------------- INPUT ---------------- private static BigDecimal readNumber(String prompt) { while (true) { System.out.print(prompt); if (!SCANNER.hasNextLine()) { return null; } String input = SCANNER.nextLine().trim(); if (input.equalsIgnoreCase("exit")) { return null; } if (input.isEmpty()) { System.out.println("Input cannot be empty."); continue; } try { return new BigDecimal(input); } catch (NumberFormatException e) { System.out.println("Invalid number. Try again."); } } } // ---------------- BINARY OPERATIONS ---------------- private static void binaryOperation(String operator) { String firstPrompt = operator.equals("percent") ? "Enter base value: " : "Enter first number: "; BigDecimal a = readNumber(firstPrompt); if (a == null) { return; } String secondPrompt = operator.equals("percent") ? "Enter percentage: " : "Enter second number: "; BigDecimal b = readNumber(secondPrompt); if (b == null) { return; } BigDecimal result = calculate(operator, a, b); printResult(a + " " + operator + " " + b, result); } private static BigDecimal calculate( String operator, BigDecimal a, BigDecimal b) { switch (operator) { case "+": return a.add(b); case "-": return a.subtract(b); case "*": return a.multiply(b); case "/": if (b.signum() == 0) { throw new ArithmeticException("Division by zero."); } return a.divide(b, MC); case "%": if (b.signum() == 0) { throw new ArithmeticException("Modulus by zero."); } return a.remainder(b); case "percent": return a.multiply(b).divide(BigDecimal.valueOf(100), MC); case "^": return power(a, b); default: throw new IllegalArgumentException("Unknown operation."); } } // ---------------- POWER ---------------- private static BigDecimal power(BigDecimal base, BigDecimal exponent) { if (exponent.stripTrailingZeros().scale() <= 0) { final int n; try { n = exponent.intValueExact(); } catch (ArithmeticException e) { throw new ArithmeticException( "Integer exponent is outside the supported range."); } if (Math.abs((long) n) > MAX_INTEGER_POWER) { throw new ArithmeticException( "Integer exponent must be between -10000 and 10000."); } if (n < 0 && base.signum() == 0) { throw new ArithmeticException( "Zero cannot have a negative exponent."); } if (n >= 0) { return base.pow(n); } return BigDecimal.ONE.divide(base.pow(-n), MC); } if (base.signum() < 0) { throw new ArithmeticException( "Fractional powers of negative bases are not supported."); } if (base.signum() == 0 && exponent.signum() < 0) { throw new ArithmeticException( "Zero cannot have a negative exponent."); } double result = Math.pow( base.doubleValue(), exponent.doubleValue()); if (!Double.isFinite(result)) { throw new ArithmeticException("Power result is outside the supported range."); } return BigDecimal.valueOf(result); } // ---------------- UNARY / SCIENTIFIC OPERATIONS ---------------- private static void unaryOperation(String operation) { BigDecimal number = readNumber("Enter number: "); if (number == null) { return; } BigDecimal result; switch (operation) { case "sqrt": if (number.signum() < 0) { throw new ArithmeticException( "Square root of a negative number is not real."); } result = number.sqrt(MC); break; case "factorial": result = factorial(number); break; case "sin": result = trig(number, "sin"); break; case "cos": result = trig(number, "cos"); break; case "tan": result = trig(number, "tan"); break; case "log": if (number.signum() <= 0) { throw new ArithmeticException( "Logarithm requires a positive number."); } result = finiteResult(Math.log10(number.doubleValue())); break; case "ln": if (number.signum() <= 0) { throw new ArithmeticException( "Natural logarithm requires a positive number."); } result = finiteResult(Math.log(number.doubleValue())); break; case "exp": result = finiteResult(Math.exp(number.doubleValue())); break; case "abs": result = number.abs(); break; default: throw new IllegalArgumentException("Unknown operation."); } printResult(operation + "(" + number + ")", result); } private static BigDecimal trig(BigDecimal number, String operation) { double angle = number.doubleValue(); if (!Double.isFinite(angle)) { throw new ArithmeticException("Angle is outside the supported range."); } if (degreeMode) { angle = Math.toRadians(angle); } double result; switch (operation) { case "sin": result = Math.sin(angle); break; case "cos": result = Math.cos(angle); break; case "tan": // Detect angles at which tangent is mathematically undefined. double normalized = degreeMode ? number.remainder(BigDecimal.valueOf(180)).doubleValue() : Math.toDegrees(angle) % 180.0; if (Math.abs(Math.abs(normalized) - 90.0) < 1e-12) { throw new ArithmeticException( "Tangent is undefined at this angle."); } result = Math.tan(angle); break; default: throw new IllegalArgumentException("Unknown trigonometric function."); } return finiteResult(result); } private static BigDecimal factorial(BigDecimal number) { if (number.signum() < 0 || number.stripTrailingZeros().scale() > 0) { throw new ArithmeticException( "Factorial requires a non-negative integer."); } if (number.compareTo(BigDecimal.valueOf(MAX_FACTORIAL)) > 0) { throw new ArithmeticException( "Maximum factorial input is " + MAX_FACTORIAL + "."); } int n = number.intValueExact(); BigInteger result = BigInteger.ONE; for (int i = 2; i <= n; i++) { result = result.multiply(BigInteger.valueOf(i)); } return new BigDecimal(result); } // Converts a finite double result to BigDecimal. private static BigDecimal finiteResult(double value) { if (!Double.isFinite(value)) { throw new ArithmeticException( "Result is outside the supported numeric range."); } return BigDecimal.valueOf(value); } // ---------------- MEMORY FUNCTIONS ---------------- private static void storeMemory() { BigDecimal value = readNumber("Enter value to store: "); if (value != null) { memory = value; System.out.println("Stored: " + format(memory)); } } private static void recallMemory() { System.out.println("Memory value: " + format(memory)); addHistory("MR = " + format(memory)); } private static void addToMemory() { BigDecimal value = readNumber("Enter value to add to memory: "); if (value != null) { memory = memory.add(value); System.out.println("Updated memory: " + format(memory)); } } // ---------------- HISTORY AND RESULTS ---------------- private static void printResult(String expression, BigDecimal result) { String formatted = format(result); System.out.println("\nExpression: " + expression); System.out.println("Result: " + formatted); addHistory(expression + " = " + formatted); } private static void addHistory(String entry) { HISTORY.add(entry); } private static void showHistory() { System.out.println("\n--------- CALCULATION HISTORY ---------"); if (HISTORY.isEmpty()) { System.out.println("No calculations yet."); return; } for (int i = 0; i < HISTORY.size(); i++) { System.out.println((i + 1) + ". " + HISTORY.get(i)); } System.out.println("---------------------------------------"); } private static String format(BigDecimal number) { if (number.signum() == 0) { return "0"; } return number.stripTrailingZeros().toPlainString(); } }
